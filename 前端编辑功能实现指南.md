### 3.1 会员编辑表单组件 (MemberEditForm.vue)

```vue
<template>
  <div class="member-edit-form">
    <h2>{{ title }}</h2>
    
    <el-form :model="form" :rules="rules" ref="formRef" label-width="120px">
      <!-- 基本信息（所有用户可编辑） -->
      <el-form-item label="用户名" prop="name">
        <el-input v-model="form.name" placeholder="请输入用户名" />
      </el-form-item>

      <el-form-item label="邮箱" prop="email">
        <el-input v-model="form.email" placeholder="请输入邮箱" type="email" />
      </el-form-item>

      <el-form-item label="手机号" prop="phone">
        <el-input v-model="form.phone" placeholder="请输入手机号" />
      </el-form-item>

      <el-form-item label="地址" prop="address">
        <el-input v-model="form.address" placeholder="请输入地址" type="textarea" />
      </el-form-item>

      <el-form-item label="入会日期" prop="join_date">
        <el-date-picker
          v-model="form.join_date"
          type="date"
          placeholder="选择入会日期"
          format="YYYY-MM-DD"
          value-format="YYYY-MM-DD"
        />
      </el-form-item>

      <!-- 管理员专属字段 -->
      <template v-if="isAdmin">
        <el-form-item label="状态" prop="status">
          <el-select v-model="form.status" placeholder="选择状态">
            <el-option label="活跃" value="active" />
            <el-option label="禁用" value="inactive" />
          </el-select>
        </el-form-item>

        <el-form-item label="最大借阅限制" prop="max_borrow_limit">
          <el-input-number v-model="form.max_borrow_limit" :min="1" :max="20" placeholder="设置最大借阅数" />
        </el-form-item>
      </template>

      <el-form-item>
        <el-button type="primary" @click="submitForm" :loading="isLoading">提交</el-button>
        <el-button @click="resetForm">重置</el-button>
        <el-button @click="cancelEdit">取消</el-button>
      </el-form-item>
    </el-form>
  </div>
</template>

<script setup>
import { ref, reactive, computed, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { useStore } from 'vuex'
import axios from 'axios'

// 组件属性
const props = defineProps({
  memberId: {
    type: Number,
    required: true
  },
  isEdit: {
    type: Boolean,
    default: true
  }
})

// 事件
const emit = defineEmits(['update:modelValue', 'success', 'cancel'])

// Vuex Store
const store = useStore()

// 表单引用
const formRef = ref()
const isLoading = ref(false)

// 当前用户角色
const isAdmin = computed(() => {
  return store.state.user?.role === 'admin'
})

// 表单标题
const title = computed(() => {
  return props.isEdit ? '编辑会员信息' : '添加会员'
})

// 表单数据
const form = reactive({
  name: '',
  email: '',
  phone: '',
  address: '',
  join_date: '',
  status: 'active',
  max_borrow_limit: 5
})

// 表单验证规则
const rules = reactive({
  name: [
    { required: true, message: '请输入用户名', trigger: 'blur' },
    { min: 2, max: 50, message: '长度在 2 到 50 个字符', trigger: 'blur' }
  ],
  email: [
    { type: 'email', message: '请输入正确的邮箱地址', trigger: ['blur', 'change'] }
  ],
  phone: [
    { pattern: /^1[3-9]\d{9}$/, message: '请输入正确的手机号码', trigger: 'blur' }
  ],
  join_date: [
    { required: true, message: '请选择入会日期', trigger: 'change' }
  ]
})

// 加载会员数据
const loadMemberData = async () => {
  if (!props.isEdit) return
  
  try {
    const response = await axios.get(`/api/members/${props.memberId}`, {
      headers: {
        Authorization: `Bearer ${store.state.token}`
      }
    })
    
    const memberData = response.data
    // 填充表单数据
    Object.assign(form, {
      name: memberData.name,
      email: memberData.email || '',
      phone: memberData.phone || '',
      address: memberData.address || '',
      join_date: memberData.join_date,
      status: memberData.status,
      max_borrow_limit: memberData.max_borrow_limit
    })
  } catch (error) {
    ElMessage.error('加载会员信息失败：' + (error.response?.data?.error || error.message))
    emit('cancel')
  }
}

// 提交表单
const submitForm = async () => {
  if (!formRef.value) return
  
  try {
    await formRef.value.validate()
    isLoading.value = true
    
    // 构建请求数据
    const requestData = { ...form }
    
    // 发送API请求
    const response = await axios.put(`/api/members/${props.memberId}`, requestData, {
      headers: {
        Authorization: `Bearer ${store.state.token}`
      }
    })
    
    ElMessage.success('会员信息更新成功')
    emit('success', response.data)
  } catch (error) {
    if (error.response) {
      // API错误响应
      ElMessage.error('更新失败：' + error.response.data.error)
    } else if (error.name === 'Error') {
      // 表单验证错误
      ElMessage.error('请检查表单填写是否正确')
    } else {
      // 其他错误
      ElMessage.error('更新失败：' + error.message)
    }
  } finally {
    isLoading.value = false
  }
}

// 重置表单
const resetForm = () => {
  if (formRef.value) {
    formRef.value.resetFields()
  }
}

// 取消编辑
const cancelEdit = () => {
  emit('cancel')
}

// 组件挂载时加载数据
onMounted(() => {
  loadMemberData()
})
</script>

  max-width: 800px;
  margin: 0 auto;
  padding: 20px;
}

.el-form-item {
  margin-bottom: 20px;
}
</style>
```

## 4. API错误处理指南

### 4.1 常见错误类型

| 错误信息 | 处理方式 |
|---------|---------|
| 用户名已存在 | 提示用户更换用户名 |
| 邮箱已存在 | 提示用户更换邮箱 |
| 入会日期格式错误 | 确保日期选择器格式正确 |
| Missing Authorization Header | 跳转到登录页面 |
| Insufficient permissions | 提示用户权限不足 |
| 404 Not Found | 提示会员不存在，刷新列表 |

### 4.2 统一错误处理

在axios拦截器中实现统一的错误处理：

```javascript
// src/utils/request.js
import axios from 'axios'
import { ElMessage, ElMessageBox } from 'element-plus'
import store from '@/store'

// 创建axios实例
const service = axios.create({
  baseURL: process.env.VUE_APP_API_BASE_URL,
  timeout: 10000
})

// 请求拦截器
service.interceptors.request.use(
  config => {
    // 添加token
    if (store.state.token) {
      config.headers.Authorization = `Bearer ${store.state.token}`
    }
    return config
  },
  error => {
    return Promise.reject(error)
  }
)

// 响应拦截器
service.interceptors.response.use(
  response => {
    return response
  },
  error => {
    const { response } = error
    
    if (response) {
      // 有响应体
      switch (response.status) {
        case 401:
          // 未授权
          ElMessageBox.confirm(
            '登录已过期，请重新登录',
            '提示',
            {
              confirmButtonText: '重新登录',
              cancelButtonText: '取消',
              type: 'warning'
            }
          ).then(() => {
            store.dispatch('logout')
            window.location.href = '/login'
          })
          break
        case 403:
          // 权限不足
          ElMessage.error('操作权限不足')
          break
        case 404:
          // 资源不存在
          ElMessage.error('请求的资源不存在')
          break
        case 400:
          // 请求参数错误
          ElMessage.error(response.data.error || '请求参数错误')
          break
        default:
          // 其他错误
          ElMessage.error(response.data.error || '服务器错误')
      }
    } else {
      // 无响应体（网络错误等）
      ElMessage.error('网络错误，请检查您的网络连接')
    }
    
    return Promise.reject(error)
  }
)

export default service
```
